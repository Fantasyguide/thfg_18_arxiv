// =====================================================================================
// 109 合成卡的「**伪 boss 载体**」（card-expand id 109，花映塚 × 獣王園）—— C3/C4 的 ECL 侧
//
// ## owner 的口径（2026-10-01）
//
// 「这个 boss 是**伪 boss**，只是我给我自己加特效，**其他什么 boss 标识都别干**」
// 「最好是**除了透明度之外完全复刻符卡**，**包括对于敌机的「自机狙」**等等」
// 「这 4 个人应该没有位移符卡——以 boss 自身的体术判定作为符卡一部分的符卡」
// ⇒ 本子程序**只做两件事**：① 在版底锚点上**隐形站着**；② 按 `plNN.ecl` 的节奏**发敌弹**。
//   **不做**：不设 boss 标志、不挂血条/名字、不进 boss 表、不做出场特效、不用体术判定
//   （⇒ `setHurtbox/setHitbox` 收到最小，**不挡自机**）。
//
// ## 为什么是「ECL 敌弹」而不是 sht 自机弹（owner 定的混合路线）
//
// 一手：TH18 的 ECL **只能驱动敌弹**（`etNew`(600) 族进敌弹池；242 条 opcode 里**没有**
//   玩家弹 opcode —— 见 `TUNING.md` §2.0v-7 的架构结论）。而「**完全复刻符卡**」要求 ECL
//   （TH09 的符卡本体就是 ECL 敌弹）。
// ⇒ C1/C2 留 sht（自机弹：原生伤敌机 + 自机狙打敌机）；**C3/C4 走这里**（ECL 敌弹 + 卡侧自伤源）。
//
// ## 一手的转写依据（源 = TH09 `plNN.ecl`，目标 = TH18 `etNew` 族）
//
// 源规格（我解的，见 `engine/ecl/th09/01-et-family-args.md`）：TH09 的 `et*`（opcode 96–104）
// 是「一条指令带全部参数」，九个实参的落点已逐个坐实：
//
//   | TH09 | 语义 | TH18 |
//   | arg1 / arg2 | (弹型, 子型) | `etSprite(id, arg1, arg2)` |
//   | arg3 / arg4 | (路数, 波数) | `etCount(id, arg3, arg4)` |
//   | arg5 / arg6 | (速度, 速度2) | `etSpeed(id, arg5, arg6)` |
//   | arg7 / arg8 | (基准角, 层间偏角) | `etAngle(id, arg7, arg8)` |
//   | opcode − 96 | 模式 | `etAim(id, mode)` |
//   | arg9 | 标志位 | 🟡 暂无对应（TH18 没有这个开关）⇒ **丢掉**（登记在 TUNING）|
//
// 目标语法（一手：原生 `default.ecl` 的反编译原文，`native/build/ecl/_ufo_tmp/default.retail.txt`
// `0x52` 行起）：
//
//   `etNew(0); etAim(0, 0); etSprite(0, 7, 2); etCount(0, 1, 1);`
//   `etAngle(0, 0.0f, 0.05235988f); etSpeed(0, %F0, 1.0f); … etOn(0);`
//
// ## 本子程序 = 灵梦 `pl00.ecl` 的 `Sub0` 逐条转写（其余角色 🟡 占位）
//
// 一手 `pl00.ecl` 的 `Sub0_572`（每 40 帧一轮，四个环）：
//
//   `etRing    (2,  2, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 514);`
//   `etRingAway(2, 15, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 642);`
//   `etRingAway(0,  2, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 514);`
//   `etRing    (0, 15, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 642);`
//
//   `[10000] = [10101] * 2`（`[10101]` ∈ {1,2,3} ⇒ 路数 2/4/6）；
//   `[10017.0f] = 1.0f + [10101] * 0.15f`（速度随档涨）；`[10016.0f]` = 一个位置/角度变量。
//
// ⚠ **两处「因为版底锚点而必须改」的适配（owner 的 180° 反转模型）**：
//   1. 一手是「boss 在对手场地上方」⇒ 基准角是**朝下**的；我们把场地 180° 反转搬到版底 ⇒
//      基准角取 **`-1.5707964f`（正上）**，即 `PPB_BASE` 常量（卡侧锚点也是这么摆的）。
//   2. 一手 `etRingAway`（TH09 模式 5）在 TH18 没有同名模式（TH16/18 的 5 = 「固定弹微调」）
//      ⇒ 本文件用 **`etAim 3`（固定弹环形）+ 基准角 + π** 表达「反向的环」，🟡 登记为待实跑校。
//
// ## 状态通道（与 `spirit109` 同一套，卡侧 ↔ ECL）
//
//   `%F2/%F3`（float，`enemy+0x14d0/+0x14d4`）= 本帧目标坐标（卡侧每帧写版底锚点）。
//   `$I0`（int）= 状态：`0` = 正常发弹；`999` = 卡侧喊走（自删）。
//   `$I1`（int）= **档位**（1 = C1 … 4 = C4；卡侧写）—— 决定路数与速度（照一手的 `[10101]`）。
//
// ## 工具链
//
//   合并：`assets/ecl/make_ufo_ecl.py` 把本文件与 `ufo.ecl.txt` / `green_extra.ecl.txt` /
//         `spirit109.ecl.txt` 一起合进原生 `default.ecl` → `native/build/ecl/default.ecl`
//         （`make ecl-ufo` 触发）。
//   子程序名 `CEPofvBoss109` 必须**不与原生 `default.ecl` 重名**（合并脚本会当场报错）。
// =====================================================================================

void CEPofvBoss109()
{
    anmSelect(2);
    anmSetSprite(0, -1);            // 本体隐形（原版「拆槽」，照 spirit109）
    setHurtbox(1.0f, 1.0f);         // ★ 伪 boss：判定收到最小 ⇒ 不挡自机、也没有体术判定
    setHitbox(1.0f, 1.0f);

CEPofvBoss109_loop:
    unless ($I0 != 999) goto CEPofvBoss109_leave @ 0;
    movePosTime(1, 1, %F2, %F3);    // 位置由卡侧每帧写到版底锚点
    // ★ 一手的 `[10000] = [10101] * 2`（路数）与 `[10017.0f] = 1.0f + [10101] * 0.15f`（速度）：
    //   本卡把「档位」$I1（1..4）当一手的 `[10101]` 用 ⇒ 路数 = 2*档、速度 = 1.0 + 0.15*档。
    setInt($I2, $I1);
    mathIntMul($I2, $I2, 2);        // $I2 = 路数 = 2 × 档
    setFloat(%F0, 1.0f);
    mathFloatMul(%F1, $I2, 0.075f); // 0.15 × 档 ≈ 0.075 × 路数（一手：1.0 + 0.15×[10101]）
    mathFloatAdd(%F0, %F0, %F1);    // %F0 = 速度 = 1.0 + 0.15×档

    // ── 四个环（一手 `Sub0_572` 那四行的顺序不变）────────────────────────────────────
    etNew(0);
    etAim(0, 3);                    // 一手 opcode 99 = etRing ⇒ TH18 mode 3（固定弹环形）
    etSprite(0, 2, 2);              // 一手 arg1/arg2 = (2, 2)
    etCount(0, $I2, 1);             // 一手 arg3/arg4 = (路数, 1)
    etAngle(0, -1.5707964f, 0.0f);  // ★ 适配：基准角 = 正上（一手是朝下；见文件头 ⚠1）
    etSpeed(0, %F0, 0.5f);          // 一手 arg5/arg6 = (速度, 0.5f)
    etOn(0);

    etNew(1);
    etAim(1, 3);                    // 一手 opcode 101 = etRingAway ⇒ mode 3 + 基准角 +π（见文件头 ⚠2）
    etSprite(1, 2, 15);
    etCount(1, $I2, 1);
    etAngle(1, 1.5707964f, 0.0f);
    etSpeed(1, %F0, 0.5f);
    etOn(1);

    etNew(2);
    etAim(2, 3);
    etSprite(2, 0, 2);
    etCount(2, $I2, 1);
    etAngle(2, -1.5707964f, 0.0f);
    etSpeed(2, %F0, 0.5f);
    etOn(2);

    etNew(3);
    etAim(3, 3);
    etSprite(3, 0, 15);
    etCount(3, $I2, 1);
    etAngle(3, 1.5707964f, 0.0f);
    etSpeed(3, %F0, 0.5f);
    etOn(3);

+40: //40  一手：`+40: ins_5(40, Sub0_572, [10001])` = 每 40 帧一轮
    goto CEPofvBoss109_loop @ 0;

CEPofvBoss109_leave:
    delete();
    return;
}
