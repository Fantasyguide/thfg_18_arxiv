// =====================================================================================
// 109 合成卡的「**伪 boss 载体**」（card-expand id 109，花映塚 × 獣王園）—— C3/C4 的 ECL 侧
//
// ## owner 的口径（2026-10-01）
//
// 「这个 boss 是**伪 boss**，只是我给我自己加特效，**其他什么 boss 标识都别干**」
// 「最好是**除了透明度之外完全复刻符卡**，**包括对于敌机的「自机狙」**等等」
// 「这 4 个人应该没有位移符卡——以 boss 自身的体术判定作为符卡一部分的符卡」
// ⇒ 本子程序**只做两件事**：① 在版底锚点上**隐形站着**；② 按 `plNN.ecl` 的节奏**发敌弹**。
//   **不做**：不设 boss 标志、不挂血条/名字、不进 boss 表、不做出场特效、不用体术判定。
//
// ## 一手的转写依据（源 = TH09 `plNN.ecl`，目标 = TH18 `etNew` 族）
//
// 源规格（逐条坐实，见 `engine/ecl/th09/01-et-family-args.md`）：TH09 的 `et*`（opcode 96–104）
// 是「一条指令带全部参数」，九个实参的落点与语义：
//
//   | TH09 实参 | 语义 | TH18 |
//   | arg1 / arg2 | (弹型, 子型) | `etSprite(id, arg1, arg2)` |
//   | arg3 / arg4 | (路数, 波数) | `etCount(id, arg3, arg4)` |
//   | arg5 / arg6 | (速度, 速度2；每颗取 `rnd×(a−b)+b`) | `etSpeed(id, arg5, arg6)` |
//   | arg7 / arg8 | (基准角, 层间偏角) | `etAngle(id, arg7, arg8)` |
//   | opcode − 96 | 模式 | `etAim(id, mode)` |
//   | arg9 | 标志位（bit9 → `0x43e380`） | 🟡 TH18 无对应开关 ⇒ **丢掉** |
//
// ★ **模式一一对应**（这条让转写变成机械活）：TH09 的 1 `etFan` / 2 `etRingAimed` / 3 `etRing`
//   与 TH16/18 的 `etAim` **1（固定弹）/ 2（自机狙环形）/ 3（固定弹环形）同号同义**。
//   ⚠ 例外：TH09 的 **5 `etRingAway`** 在 TH18 没有同名模式（TH16/18 的 5 = 「固定弹微调」）
//     ⇒ 用 `etAim 3` + 基准角 **+π** 表达「反向的环」（🟡 待实跑校）。
//
// 目标语法（一手：原生 `default.ecl` 反编译原文）：
//   `etNew(0); etAim(0, 0); etSprite(0, 7, 2); etCount(0, 1, 1);`
//   `etAngle(0, 0.0f, 0.05235988f); etSpeed(0, %F0, 1.0f); … etOn(0);`
//
// ## 三个角色的形态（逐条转写自 `pl00/pl01/pl02.ecl`）
//
// **灵梦 `pl00`（`Sub0_572`，每 40 帧一轮四个环）**
//   `etRing    (2,  2, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 514)`
//   `etRingAway(2, 15, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 642)`
//   `etRingAway(0,  2, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 514)`
//   `etRing    (0, 15, [10000], 1, [10017.0f], 0.5f, [10016.0f], 0.0f, 642)`
//   `[10000] = [10101] × 2`（路数，`[10101]` ∈ {1,2,3}）；`[10017.0f] = 1.0 + 0.15 × [10101]`
//
// **魔理沙 `pl01`（`Sub0` / `Sub1` / `Sub4`）—— ★ 不是环，是「路数 1、波数 8」的**连续细流**
//   `etRing(12,  6, 1, 8, [10020.0f], 0.5f, [10018.0f],  0.0f, 512)`
//   `etRing(13,  6, 1, 8, [10020.0f], 0.5f, [10018.0f],  0.0f, 512)`
//   `etRing( 0,  6, 1, 8,        2.0f,   0.5f, [10018.0f],  0.0f, 512)`
//   `etRing(12, 10, 1, 8, [10020.0f], 0.5f, 2.3561945f,     0.0f, 512)`   ← 基准角 3π/4
//   `etRing(13, 10, 1, 8, [10020.0f], 0.5f, 0.7853982f,     0.0f, 512)`   ← 基准角 π/4
//   `etRing(12, 13, 1, 1, [10018.0f], 1.3f, [10082.0f],     0.0f, 546)`
//   ⇒ 观感 = **细而密的直线流 + 两股 ±45° 的斜流**（与 owner「魔理沙＝激光那一路」一致）
//
// **咲夜 `pl02`（`Sub0` / `Sub1` / `Sub3` / `Sub4` / `Sub7`）—— 扇形与环混用，层间偏角 12°**
//   `etRingAimed(20, 1, 4,      1, 0.0f,      0.5f, [10023.0f], 0.0f, 528)`  ← mode 2
//   `etRing     (20, 2, 7,      1, [10016.0f], 2.0f, [10082.0f], 0.0f, 18946)`
//   `etFan      (20, 3, [10000],1, [10016.0f], 2.0f, [10017.0f], [10019.0f], 514)`
//   `etFan      ( 0, 6, [10001],1, [10017.0f], 0.5f, [10016.0f], 0.20943952f, 514)` ← 层间偏角 12°
//   ⇒ 观感 = **会瞄人的扇形 + 一圈 + 层间错开 12° 的扇**（`0.20943952` = 12° 是铁证）
//
// ⚠ **因为「版底锚点 + 180° 反转」必须改的一处**：一手是「boss 在对手场地上方」⇒ 基准角朝**下**；
//   我们把场地 180° 反转搬到**版底** ⇒ 基准角一律取 **`-1.5707964f`（正上）**，
//   一手那些**相对**基准角（±45°、+π）照相对关系平移。
//
// ## 状态通道（与 `spirit109` 同一套，卡侧 ↔ ECL）
//
//   `%F2/%F3`（float）= 本帧目标坐标（卡侧每帧写版底锚点）。
//   `$I0`（int）= 状态：`0` = 正常发弹；`999` = 卡侧喊走（自删）。
//   `$I1`（int）= **档位** 1..4（卡侧写）—— 照一手的 `[10101]`：路数 ×2、速度 +0.15/档。
//   `$I2`（int）= **角色**：`0` = 灵梦 / `1` = 魔理沙 / `2` = 咲夜（卡侧写；其余值按灵梦跑）。
//   `$I3` = 本子程序的临时寄存器（路数）；`%F0/%F1` = 速度 / 临时。
//
// ## 工具链
//
//   合并：`assets/ecl/make_ufo_ecl.py`。子程序名 `CEPofvBoss109` 必须不与原生重名。
// =====================================================================================

void CEPofvBoss109()
{
    var R;                          // ★ 本子程序唯一的局部（帧间保持）：本档还要发几轮
                                    //   ⚠ TH18 的 ECL 全局只有 `$I0..$I3` / `%F0..%F3`（见 `engine.h`）
                                    //     ⇒ 第 5 个整数得用**局部**（`var R;` + `$R`）
    anmSelect(2);
    anmSetSprite(0, -1);            // 本体隐形（原版「拆槽」，照 spirit109）
    setHurtbox(1.0f, 1.0f);         // ★ 伪 boss：判定收到最小 ⇒ 不挡自机、也没有体术判定
    setHitbox(1.0f, 1.0f);

CEPofvBoss109_loop:
    unless ($I0 != 999) goto CEPofvBoss109_leave @ 0;
    movePosTime(1, 1, %F2, %F3);    // 位置由卡侧每帧写到版底锚点
    // 一手的 `[10000] = [10101] × 2 + 32`（路数）与 `[10017.0f] = 1.0 + 0.15 × [10101]`（速度）
    // ⇒ 本卡把「档位」$I1（1..4）当一手的 `[10101]` 用。
    // ⚠⚠ **TH18 的 ECL 是栈式的**（一手写法见 `assets/ecl/ufo.ecl.txt`：
    //   `%A; 6; addf(); setf(%A);`）—— 我第一版写成 `mathIntMul($I3, $I1, 2)` 那种「函数式」，
    //   而 `thecl` 对**未知助记符只 warning**（编成「子程序调用」）⇒ 那几行**根本没算数**
    //   （`@mathIntMul(...)`）。已按栈式重写，并给构建加了硬闸（见 `make_ufo_ecl.py`）。
    $I1; 2; muli(); 32; addi(); seti($I3);           // $I3 = 路数 = $I1 × 2 + 32
    $I1; 0.15f; mulf(); 1.0f; addf(); setf(%F0);     // %F0 = 速度 = 1.0 + 0.15 × $I1
    // ★ 一手的阶段结构：`loop(40, Sub0_572, [10001])` = 整轮**重复 `[10001]` 次**（1/2/3）
    //   ⇒ 本卡把重复次数也取成档位（与一手的 `[10001]` 同源：那三个分支就是按 `[10101]` 分的）。
    $R = $I1;                                        // $R = 本档还要发几轮
    // ⚠ 全局变量（`$I*`）必须用**赋值式**写（`$I4 = $I1;`）—— `$I1; seti($I4);` 那种
    //   写法会被编到**局部** `$A` 上（反编译看得见），而且往返自洽会当场报
    //   「variable not found: A」（本轮就是被它抓到的）。

CEPofvBoss109_volley:
    // ── 角色分派（$I2），每 40 帧一轮（一手：灵梦 `+40:`、魔理沙也是 40 帧节拍）──────
    unless ($I2 != 1) goto CEPofvBoss109_marisa @ 0;
    unless ($I2 != 2) goto CEPofvBoss109_sakuya @ 0;

    // ═══ 灵梦 `pl00`：四个环（mode 3 / 5；`etRingAway` = mode 3 + 基准角 +π）═══
    etNew(0);
    etAim(0, 3);
    etSprite(0, 2, 2);
    etCount(0, $I3, 1);
    etAngle(0, -1.5707964f, 0.0f);
    etSpeed(0, %F0, 0.5f);
    etOn(0);

    etNew(1);
    etAim(1, 3);
    etSprite(1, 2, 15);
    etCount(1, $I3, 1);
    etAngle(1, 1.5707964f, 0.0f);   // 一手 etRingAway ⇒ 基准角 +π
    etSpeed(1, %F0, 0.5f);
    etOn(1);

    etNew(2);
    etAim(2, 3);
    etSprite(2, 0, 2);
    etCount(2, $I3, 1);
    etAngle(2, -1.5707964f, 0.0f);
    etSpeed(2, %F0, 0.5f);
    etOn(2);

    etNew(3);
    etAim(3, 3);
    etSprite(3, 0, 15);
    etCount(3, $I3, 1);
    etAngle(3, 1.5707964f, 0.0f);
    etSpeed(3, %F0, 0.5f);
    etOn(3);
    goto CEPofvBoss109_next @ 0;

CEPofvBoss109_marisa:
    // ═══ 魔理沙 `pl01`：路数 1、波数 8 的细流（三种弹型）+ 两股 ±45° 斜流 + 一发特弹 ═══
    etNew(0);
    etAim(0, 3);
    etSprite(0, 12, 6);
    etCount(0, 1, 8);               // 一手：路数 1、波数 8 ⇒ 连续 8 连发
    etAngle(0, -1.5707964f, 0.0f);
    etSpeed(0, %F0, 0.5f);
    etOn(0);

    etNew(1);
    etAim(1, 3);
    etSprite(1, 13, 6);
    etCount(1, 1, 8);
    etAngle(1, -1.5707964f, 0.0f);
    etSpeed(1, %F0, 0.5f);
    etOn(1);

    etNew(2);
    etAim(2, 3);
    etSprite(2, 0, 6);
    etCount(2, 1, 8);
    etAngle(2, -1.5707964f, 0.0f);
    etSpeed(2, 2.0f, 0.5f);         // 一手这一支的速度是定值 2.0f
    etOn(2);

    etNew(3);
    etAim(3, 3);
    etSprite(3, 12, 10);
    etCount(3, 1, 8);
    etAngle(3, -2.3561945f, 0.0f);  // 一手 3π/4（相对「朝前」+45°）⇒ 平移到「正上」的左上
    etSpeed(3, %F0, 0.5f);
    etOn(3);

    etNew(4);
    etAim(4, 3);
    etSprite(4, 13, 10);
    etCount(4, 1, 8);
    etAngle(4, -0.7853982f, 0.0f);  // 一手 π/4 ⇒ 右上
    etSpeed(4, %F0, 0.5f);
    etOn(4);

    etNew(5);
    etAim(5, 3);
    etSprite(5, 12, 13);
    etCount(5, 1, 1);               // 一手 Sub4：路数 1、波数 1、速度 1.3f
    etAngle(5, -1.5707964f, 0.0f);
    etSpeed(5, 1.3f, 0.5f);
    etOn(5);
    goto CEPofvBoss109_next @ 0;

CEPofvBoss109_sakuya:
    // ═══ 咲夜 `pl02`：会瞄人的扇形 + 一圈 + 层间错开 12° 的扇（0.20943952f = 12° 是一手铁证）═══
    etNew(0);
    etAim(0, 2);                    // 一手 opcode 98 = etRingAimed（自机狙环形）⇒ TH18 mode 2 同义
    etSprite(0, 20, 1);
    etCount(0, 4, 1);
    etAngle(0, -1.5707964f, 0.0f);
    etSpeed(0, %F0, 0.5f);
    etOn(0);

    etNew(1);
    etAim(1, 3);                    // 一手 opcode 99 = etRing
    etSprite(1, 20, 2);
    etCount(1, 7, 1);
    etAngle(1, -1.5707964f, 0.0f);
    etSpeed(1, %F0, 2.0f);          // 一手 Sub3：速度2 = 2.0f
    etOn(1);

    etNew(2);
    etAim(2, 1);                    // 一手 opcode 97 = etFan（固定弹扇形）⇒ TH18 mode 1 同义
    etSprite(2, 20, 3);
    etCount(2, $I3, 1);
    etAngle(2, -1.5707964f, 0.20943952f);   // ★ 层间偏角 12°（一手 Sub7 的实参）
    etSpeed(2, %F0, 2.0f);
    etOn(2);

    etNew(3);
    etAim(3, 1);
    etSprite(3, 0, 6);
    etCount(3, $I3, 1);
    etAngle(3, -1.5707964f, 0.20943952f);
    etSpeed(3, %F0, 0.5f);
    etOn(3);
    goto CEPofvBoss109_next @ 0;

CEPofvBoss109_next:
+40: //40  一手：`+40: loop(40, Sub0_572, [10001])` —— 等 40 帧，然后把这一轮**重复 `[10001]` 次**
    $R = $R - 1;                                     // $R -= 1
    if ($R > 0) goto CEPofvBoss109_volley @ 0;       // 还有次数 ⇒ 再发一轮
    goto CEPofvBoss109_loop @ 0;                     // 这一档发完 ⇒ 回主循环（继续跟锚点/看 $I0）

CEPofvBoss109_leave:
    delete();
    return;
}
