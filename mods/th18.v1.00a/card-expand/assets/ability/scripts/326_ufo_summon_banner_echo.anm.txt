script script326 {
    // 112 星莲船 —— 大横幅的**残影**（一手：TH12 `front.anm` **script84** 逐条移植）
    //   由脚本 324 在 `+35 / +40 / +45` 各起一只（原文一模一样的三句 `scriptNew(script84)`）
    //   ⇒ 主横幅之外再有**三只同图残影**先后放大淡出（= 一道"荡开"的层次感）。
    //
    // **原文**：
    //   `sprite(sprite68); layer(6); ins_85(1); randMode(1); type(2); blendMode(1); alpha(128);
    //    scaleTime(30, 2, 2, 2); alphaTime(30, 0, 0); +30: delete();`
    //   ⚠ **原文没有 `pos()`** ⇒ 它吃的是"父 VM 的 entity_pos + 自己的 pos(0,0)"。
    //      本卡先例（脚本 232 血条投影同样不设 pos、随父走，实跑验过）⇒ 照抄"不写 pos"。
    //   中断 7/8/9/10 与 script82 **不同**：这里 **8 = 绿**(`128,255,128`)、**9 = 蓝**(`128,128,255`)
    //   —— 原文就是这样（82 是 8 = 蓝、9 = 绿）。**逐条照抄，别"顺手统一"**。
    //
    // **移植改动**：Δ2 删 `ins_85(1)`；Δ3 删 `type(2)`（理由见脚本 324 的 Δ3）；层 6 原值保留。
    sprite(@SUMMON_BANNER@);
    layer(6);
    randMode(1);
    blendMode(1);
    alpha(128);
    /* ★ owner 口径：×1/3（一手是 1.0 → 2.0）。⚠ `scale` 必须在 `scaleTime` **之前** ——
     *   anm 是顺序执行：先 `scaleTime`（起一段 30 帧插值）再 `scale` 会当场被后面的值盖掉。*/
    scale(0.333f, 0.333f);
    scaleTime(30, 2, 0.667f, 0.667f);
    alphaTime(30, 0, 0);
+30: // 30
    delete();
    interruptLabel(7);
    color(255, 128, 128);
    caseReturn();
    interruptLabel(8);
    color(128, 255, 128);
    caseReturn();
    interruptLabel(9);
    color(128, 128, 255);
    caseReturn();
    interruptLabel(10);
    color(255, 255, 128);
    colorTime(30, 0, 128, 255, 255);
    caseReturn();
