script script397 {
    // 伟大之作（id 92）发动 —— **圆形扩张的加法金色闪光 → 顶点纯白 → 全屏白**（owner 2026-10-03 时序）：
    //   「威光的光线最先出来，**几乎同时**圆盘长出来开始，但是由于慢所以一开始相对注意不到」
    //   「**先慢再快**，同时在顶点的时候加法和乘法都变成**纯白色**」
    //   「顶点也是**慢慢**变成纯白色，不是瞬间变成纯白色」「然后**全屏死亡**」。
    // 实心盘 = `sprite(-1)` + `drawPoly(1.0f,32)`（半径 = scale；原生 script49 同配方）。
    //   ① 0→24 帧：0 → 90 半径（**慢** —— 一开始几乎看不出）；
    //   ② 24→34 帧：90 → 640 半径（**快** ⇒ 罩住整个游玩区）；
    //   ③ alpha 从 0 **线性**升到 255（走满 34 帧 ⇒ 顶部自然到纯白，不是跳变）；color 在最后 10 帧
    //      渐变到 (255,255,255) ⇒ **顶点 = 纯白 + 满 alpha = 全屏白（"全屏死亡"）**；
    //   ④ 顶点停留 8 帧，再 20 帧退 0 自毁。
-1:
    layer(19);
    sprite(-1);
    zWriteDisable(1);
    blendMode(1);
    drawPoly(1.0f, 32);
    color(255, 206, 110);
    alpha(0);
    scale(0.0f, 0.0f);
0:
    scaleTime(42, 0, 90.0f, 90.0f);      // 慢段：42 帧才长到 90 半径
    alphaTime(60, 0, 255);               // α 一路线性升到顶（第 60 帧 = 1 s = 顶点）
+42:
    scaleTime(18, 0, 640.0f, 640.0f);    // 快段：18 帧冲到 640（罩满游玩区）
    colorTime(18, 0, 255, 255, 255);     // 顶点 = 纯白
+18:
    scaleTime(8, 0, 640.0f, 640.0f);     // 顶点停留（保持 640 的"空转" ramp —— 比空标签更稳）
+8:
    // ★ owner 2026-10-03：「然后是**一个白色相加的缓出**」—— 峰值之后这一层用**缓出**收：
    //   两段直线近似 ease-out（先快后慢的长尾）：6 帧从 255 掉到 60，再用 30 帧拖回 0。
    alphaTime(6, 0, 60);
+6:
    alphaTime(30, 0, 0);
+30:
    delete();
}
