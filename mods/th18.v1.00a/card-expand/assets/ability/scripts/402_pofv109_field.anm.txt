    //
    //   ```asm
    //   sprite(-1); layer(28);
    //   blendMode(2);                          ← ★ 减去混合
    //   drawPoly(1.0f, 12);                    ← ★ **12 边形**（不是圆）
    //   color(128,128,255); color2(64,64,64);  ← 外/内两色（蓝 + 暗灰）
    //   angleVel(0.017453292f, 0.0f, 0.017453292f);  ← ★ **每帧自转 1°**
    //   alpha(200); alpha2(40);                ← 内实、外淡
    //   scale(13.0f, 13.0f);                   ← ★ 半径约 **13 px**（**很小**，不是大场）
    //   offset152: scaleTime(20, 0, 14.0f, 14.0f);   ← 13 → 14（20 帧）
    //   +20:       scaleTime(20, 0, 12.0f, 12.0f);   ← 14 → 12（20 帧）
    //   +20: jmp(offset152, 0);                ← ★ **循环** ⇒ **呼吸**（不是单向扩张 ✗）
    //   case(1): alphaTime(10,0,0); alpha2Time(10,0,0); +10: delete();   ← 收尾淡出
    //   ```
    //
    //   ★★ **这就是「扩张的感觉不对」的根因**：一手**根本不扩张** ✗ ——
    //     它是**固定大小（半径 ~13）、13↔14↔12 来回呼吸 + 缓慢自转**的一小片 ✓✓。
    //
    //   ⚠ 与旧版的关系：旧版有两稿都不对 —— ①「程序化扩张圆盘」（我自造 ✗）；

    //     ⇒ 现在有了 TH19 的原配方，**两者都不用** ✓。
    //   ⚠ 因此**卡侧不再每帧写 scale** ✗（一写就把呼吸盖掉了）——大小由本脚本自己管 ✓；

-1:
    layer(28);
    blendMode(2);
    type(1);
0:
    sprite(-1);
    drawPoly(1.0f, 12);
    color(128, 128, 255);
    color2(64, 64, 64);
    angleVel(0.017453292f, 0.0f, 0.017453292f);
    alpha(200);
    alpha2(40);
    scale(13.0f, 13.0f);

    scaleTime(20, 0, 14.0f, 14.0f);
+20: // 20
    scaleTime(20, 0, 12.0f, 12.0f);
+20: // 40

    interruptLabel(1);
    alphaTime(10, 0, 0);
    alpha2Time(10, 0, 0);
+10:
    delete();
