script script345 {
    // 112 星莲船 —— **满 B 掉落物的卡面层**（一手：TH18 卡道具在场地里的显示配方 `abcard.anm` script15）
    //
    // ★★★ 2026-09-24 第 22 轮（owner 终局口径）：「**我要和 B 碎一样大**」「**它得在各方面 == B 碎卡**」
    //   ⇒ 本层**逐条等于**原生场地卡道具的那一层 —— 一手 `abcard.anm` **script15**：
    //       `sprite(sprite11); ins_431(1); randMode(1); pos(0,0,0); layer(16);
    //        color(255,255,255); scale(0.3f, 0.3f); type(1); resampleMode(0); stop();`
    //     其中 `sprite11` = entry11 `ability/EXTEND2_min.png`（命碎卡）—— **引擎按卡 id 换成对应那张**：
    //       B 碎卡用 `BOMB2_min.png`（64×80）、整颗 B 用 **`BOMB_max.png`（256×320，自带黑框）**。
    //
    //   ⇒ 「一样大」= **屏显尺寸对齐**：`_max` 正好是 `_min` 的 **4 倍画布** ⇒ 倍率写 **0.075**（不是 0.3），
    //     两条路都落在 **19.2×24**（owner 场里 B 碎卡的尺寸）。贴图成员 `DROP_BOMB` = 原生
    //     `BOMB_max.png` sprite8 (0,0,256,320) 的 **1:1 切图**（见 `assets/ability/make_ufo_drop_art.py`）。
    //
    // **Δ（两条，都不改外观）**：
    //   · `ins_431(1)`（v8 未知 flag，一手 `engine/anm/th16/07-general.md` §8 标 ❓）本卡不写 ——
    //     本仓其它原生移植脚本同款处理（写不写都过；语义未知不发明）。
    //   · `pos(0,0,0)` 省略：本 VM 的 `entity_pos` 由 C 侧**逐帧**从原生道具坐标（`item+0xc2c`）写，
    //     原生那句 pos 是给引擎自己 spawn 的 VM 用的初值（`ufo.c` 的 `ufo_drops_draw_follow()`）。
    //   · ★ 第 21 轮那版本层还带 `scaleTime(7,9,0.9) ↔ (7,9,1.0)` 的呼吸 —— **删掉**：一手的呼吸只在
    //     **背光层**（script16）上，卡面层是静止的。两层一起呼吸 = 与 B 碎卡不同（owner 要「各方面 ==」）。
    sprite(@DROP_BOMB@);
    type(1);
    layer(16);                      /* 一手 script15：世界层 16（掉落物该在的层）*/
    color(255, 255, 255);
    /* ★★ 2026-09-25 第 24 轮（owner：「卡图不应该是 `BOMB_max.png` 吗？这张图自带黑框啊」）：
     *   成员换成 **`BOMB_max.png` 的 sprite8 (0,0,256,320)**（1:1 原生 texel）⇒ 倍率要跟着换成
     *   **0.075**（= 一手 0.3 ÷ 4，因为 `_max` 正好是 `_min` 的 4 倍画布）⇒ 屏显仍是 **19.2×24**
     *   = 与 B 碎卡同尺寸；而 `_max` 自带的外圈黑边 + 灰带（约 1 px）就跟着画出来了。*/
    scale(0.075f, 0.075f);          /* 一手 0.3 × (_min 64 / _max 256) = 0.075 ⇒ 屏显 19.2×24 */
    resampleMode(0);                /* ★ 一手原值（`engine/anm/th16/06-settings.md` §3.5：bit11 = 0 线性）*/
    stop();
}
