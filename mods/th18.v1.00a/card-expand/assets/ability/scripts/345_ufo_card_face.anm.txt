    //
    // ★ 2026-09-24 第二轮：从脚本 301 里**拆出来**独立一层 —— 301 现在是**父**（`sprite(-1)` +
    //   一手 `abcard.anm` script5 的 **`rotateTime` 摇摆**），两个子各管一层：
    //     · **本层（333）** = **卡面**：原生卡图 `abcard/ability/BOMB_max.png` 等比缩到 **26×32**、

    type(1);
    layer(16);
    color(255, 255, 255);
    scale(1.0f, 1.0f);
offset112:

+7: // 7
    scaleTime(7, 9, 1.0f, 1.0f);
+7: // 14
    jmp(offset112, 0);
    stop();
