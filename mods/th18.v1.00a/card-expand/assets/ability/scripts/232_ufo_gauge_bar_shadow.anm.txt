script script232 {
    // 星莲船（系统卡 112）血量条的**1 px 投影**（不是框！）—— 零售 TH12 v1.00b `front.anm`
    // **script137** 的逐条移植（`local/th12.v1.00b/anm/front/front.anm.txt`；正文**一字未改**）。
    //
    // 一手：
    //   ① 它是血条 script136 的 `scriptNew` 子脚本 —— TH12 `front.anm` script136 的第一行就是
    //      `scriptNew(script137)`（照抄在脚本 231 里）。
    //   ② **它是投影，不是外框**：`sprite(-1)` + `drawRect(32,4)` + `color(0,0,0)` + **`pos(1,1,0)`**
    //      = 同尺寸的黑色实心矩形，**向右下各偏 1 px** ⇒ 给红条压一条 1 px 黑影。
    //      「框」在 TH12 里是**另一条 VM**（script134 = `0x86`、`sprite102` = `front_ufo.png` 的 96×96）——
    //      用户口径是「**没有框**」⇒ 本卡不起 134（一手结构见 `engine/ufo/th12/01-ufo-system.md` §10 末）。
    //   ③ 层号：`layer(9)` = **零售原值**，比血条的 `layer(10)` **低 1** ⇒ 影子画在条的**下面**
    //      （零售就是靠这个相对关系把投影压在条后；同一树里 script136 写 10、script137 写 9，
    //      层号是各自脚本里写死的，**不是**「子继承父」）。9 与 10 都在 TH18 的世界层区间（3–19）里
    //      ⇒ 照抄不改（理由与 231 的层号段完全一致）。
    //   ④ **为什么影子要「一起起、一起跟」**：零售里它是**同一个进度条 VM 的子 VM**，父 VM 的
    //      `scale.x` 每帧被引擎写成 HP 比例（`th12:0x44a5d0`），影子**不**跟着缩放 ——
    //      也就是说零售的影是**恒长 32** 的。本卡照抄这条（影子只跟位置，不跟 scale），
    //      并在 `ufo.c` 里把「影子只跟 pos」写成注释（别顺手也去 set_scale）。
    //
    // 本件**改了哪一行**：无（连层号都是零售原值）。唯一要交代的是「卡侧照抄零售：影子只跟位置」。
    sprite(-1);
    drawRect(32.0f, 4.0f);
    layer(9);
    color(0, 0, 0);
    pos(1.0f, 1.0f, 0.0f);
    stop();
    interruptLabel(1);
    delete();
}
