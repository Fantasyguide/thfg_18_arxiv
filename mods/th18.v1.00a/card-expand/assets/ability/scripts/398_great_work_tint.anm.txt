script script398 {
    // 伟大之作（id 92）发动 —— **圆形扩张的乘法金色滤镜**（owner 2026-10-03：
    //   「乘法滤镜和闪光能不能变成**圆形扩张范围**？」）。
    // 与 397 同一套实心盘（`sprite(-1)` + `drawPoly(1.0f, 32)`，半径 = scale），区别在**叠法**：
    //   `blendMode(5)` = `SRC=DESTCOLOR / DEST=ZERO` ⇒ `rgb′ = src × dst`（乘法滤波）。
    //   ★ 几何遮罩是关键：`drawPoly` 只**栅格化盘内的像素** ⇒ 乘法只作用在盘内 ——
    //     金色就像一圈波从中心扩出去，盘外的画面一点不动（比全屏矩形精确得多）。
    //
    // ⚠ 乘法模式的 RGB **不吃 alpha**（TH18 的 alpha 是分离混合 `Aa′ = Sa`，见
    //   `engine/anm/th18/07-blendmode-rgb-only-alpha-is-sa.md`）⇒ **淡出只能用 `colorTime`**
    //   把乘法色从金插回**白**（乘 1 = 不变）：8 帧推到金 (255,205,110)，落定后 16 帧退回白再自毁。
    //   `alpha` 保持 255：`color_1.A == 0 && color_2.A == 0` 会让整台 VM **不画**。
-1:
    layer(19);
    sprite(-1);
    zWriteDisable(1);
    blendMode(5);
    drawPoly(1.0f, 32);
    color(255, 255, 255);
    alpha(255);
    scale(0.0f, 0.0f);
0:
    scaleTime(28, 4, 560.0f, 560.0f);
    colorTime(8, 0, 255, 205, 110);
+28:
    colorTime(16, 0, 255, 255, 255);
+16:
    delete();
}
