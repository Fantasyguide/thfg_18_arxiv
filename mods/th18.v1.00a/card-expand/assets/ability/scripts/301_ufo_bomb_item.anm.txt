script script301 {
    // 112 星莲船 —— **满 B 掉落物的外观**（★ 2026-09-24 第三轮：**一张烘死的合成图**，静态单层）
    //
    // ★★★ owner 两句话定了这一版：
    //   「**你要不然直接做一张合成图吧，发光难道还有动画效果吗？**」
    //   → 「**卡大了一倍，没有看见发光**」
    //   一手的答案（诚实记下来）：一手 `abcard.anm` **script16**（卡道具的显示配方）**确实有动画**
    //   —— `blendMode(1)` 加算 + `scaleTime(7,9,0.9) ↔ (7,9,1.0)` 的**无限呼吸**。但 owner 要的是
    //   "一张合成图" ⇒ 本版把**光晕与卡面烘进同一张贴图**（生成器 `make_ufo_drop_art.py` 的
    //   `make_drop_bomb_composite()`），脚本侧只剩一句 `sprite(...)` + 静止 `scale(1.0)`
    //   ⇒ **没有第二层、没有加算、没有动画**。
    //
    // **合成图是怎么来的**（`assets/ability/ufo_drop/DROP_BOMB.png`，34×40）
    //   · 底：原生卡道具的**发光卡形** `abcard/ability/abitem.png`（40×44，一手 = `abcard.anm`
    //     entry1 sprite1；alpha 剖面 = 核心平顶 194 + 外圈 8 px 渐隐）放大到画布，
    //     并把 alpha 乘 **2.2**（`GLOW_ALPHA_MUL`）——近合一手的加算（烘死的图没法再加算）。
    //   · 面：owner 指定的**原生卡图** `abcard/ability/BOMB_max.png`（256×320）等比缩到 **26×32**
    //     居中贴上去 ⇒ 四周露出一圈 4 px 的柔光 = "发光"。
    //   · 尺寸 26×32：原生整 B 道具的图标是 `bullet/item.png` sprite512 = **32×32**（`scale 0.5→1.0`
    //     ⇒ 屏显 32×32）；owner 判上一版 40×50"大了一倍"（面积 2000 ≈ 图标 1024 的两倍）⇒ 取高 32、4:5 等比。
    //     ⚠ 一手**卡道具美术**（type 0x10–0x13 那条）是 `_min.png` 64×80 texel、在 UI 层 mode 2 下
    //     = **32×40** 逻辑 px ⇒ 若要对齐那一条，把生成器的 `CARD_FIT` 改成 (32, 40) 即可（一处）。
    //
    // **Δ**：只有 sprite 名（`@DROP_BOMB@` = 我们的合成图成员）；层 `layer(16)` 原值保留
    //   （一手 script16 自己写的就是 16，且那是**世界层** = 掉落物该在的层）。
    sprite(@DROP_BOMB@);
    type(1);
    layer(16);                      // 一手 script16 的层（世界层：跟着道具在场里画）
    color(255, 255, 255);
    scale(1.0f, 1.0f);              // ★ 静止（不再呼吸、不再加算、不再有第二层）
    stop();
}
