script script301 {
    // 112 星莲船 —— **满 B 掉落物的卡面层**（一手：TH18 卡道具的显示配方 `abcard.anm` script16）
    //
    // ★★★ 2026-09-24 owner 四轮口径（逐句，最终形态）：
    //   「满B卡是 smjb，不会用 `…/cardart/native/ability/BOMB_max.png`？发光也不是 tm 这么发的啊」
    //   →「**你要不然直接做一张合成图吧**，发光难道还有动画效果吗？」
    //   →「**卡大了一倍，没有看见发光**」→「**别搞合成图了还是重新背光吧**」
    //   ⇒ **两层**：本脚本画**卡面**（原生卡图，正常混合、**不加算**），脚本 **331** 画**背光**
    //     （原生发光卡形贴图 `abitem.png`，**加算**）。合成图方案已作废（生成器里也删了）。
    //
    // **尺寸（owner「卡大了一倍」）**：卡面 = 原生卡图 `BOMB_max.png`（256×320）等比缩到 **26×32**。
    //   为什么 26×32：原生**整 B 道具图标** = `bullet/item.png` sprite512 = **32×32**（`scale 0.5→1.0`
    //   ⇒ 屏显 32×32）；owner 判上一版 40×50「大了一倍」（面积 2000 vs 1024 ≈ 两倍）⇒ 取高 32、4:5 等比。
    //   ⚠ 一手**卡道具美术**（type 0x10–0x13）是 `_min.png` 64×80 texel，在 UI 层 `resolutionMode(2)` 下
    //     = 32×40 逻辑 px ⇒ 要对齐那一条就把生成器的 `CARD_FIT` 改成 `(32, 40)`（一处）。
    //
    // **一手的动态**（script16 逐条）：`blendMode(1)` 加算 + `scaleTime(7,9,0.9) ↔ (7,9,1.0)` 的
    //   **无限呼吸**（7 帧一段、缓动 9）。**owner 问过"发光难道还有动画效果吗"——一手确实有**，
    //   所以本版两层都保留呼吸；要静止就把两支脚本里的 `jmp(offset112, 0)` 删掉、`scale` 固定 1.0。
    //
    // **Δ**：`sprite(sprite1)` → `sprite(@DROP_BOMB@)`（我们的卡面成员）；
    //   `layer(16)` 原值保留（一手 script16 自己写的就是 16，且那是**世界层** = 掉落物该在的层）；
    //   卡面这层**不加算**（加算会把卡图冲淡 —— owner 说的"发光不是这么发的"）。
    // ★★★ 2026-09-24 第二轮（owner）：「**整张卡片应当旋转摇摆的**」—— 一手那支**摇摆**就在
    //   `abcard.anm` **script5**（会摇摆的卡显示）里，逐条原样：
    //     `offset176: rotateTime(120, 21, -0.034906585, 0, -0.034906585);
    //      +120: rotateTime(120, 21, 0.06981317, 0, 0.06981317); +120: jmp(offset176, 0);`
    //   ⇒ 在 **-2° ↔ +4°** 之间用 120 帧（2 秒）一个来回、缓动 21 地**无限摇摆**。
    //   ⚠ 摆的是**父 VM**（本脚本没有 sprite）⇒ 子 VM（背光 331）继承同样的变换 ⇒ 两层一起摆。
    sprite(-1);                     /* 父自己不画（一手 script5 第一句就是 `sprite(-1)`）*/
    scriptNew(@SCRIPT_UFO_CARD_GLOW@);   /* ★ 背光层（原生卡框 abframe：加算 + 呼吸）*/
    scriptNew(@SCRIPT_UFO_CARD_FACE@);   /* ★ 卡面层（原生卡图，正常混合）*/
    type(1);
    layer(16);
    color(255, 255, 255);
    scale(1.0f, 1.0f);
offset176:
    rotateTime(120, 21, -0.034906585f, 0.0f, -0.034906585f);   // 一手 script5：-2°
+120: // 120
    rotateTime(120, 21, 0.06981317f, 0.0f, 0.06981317f);       // 一手 script5：+4°
+120: // 240
    jmp(offset176, 0);                                         // 一手 script5：无限摇摆
+20: // 260
    stop();
offset112:
    scaleTime(7, 9, 0.9f, 0.9f);    // 一手 script16：7 帧 → 0.9（缓动 9）
+7: // 7
    scaleTime(7, 9, 1.0f, 1.0f);    // 一手 script16：7 帧 → 1.0
+7: // 14
    jmp(offset112, 0);              // 一手 script16：无限呼吸
    stop();
}
