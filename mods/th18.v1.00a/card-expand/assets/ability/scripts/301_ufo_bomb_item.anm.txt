script script301 {
    // 112 星莲船 —— **满 B 掉落物的父 VM（整张卡：摆动 + 起两层）**
    // 一手：TH18 **场地里那张卡道具**的显示配方 `abcard.anm` **script14**（世界层 16 的那一棵）。
    //
    // ★★★ 2026-09-24 第 22 轮（owner 终局口径）：「**我要和 B 碎一样大**」「**它得在各方面 == B 碎卡**」
    //   ⇒ 本卡不再"照着一手的某一句自己搭"，而是**整棵照抄原生场地卡道具**：
    //       script14（本脚本，父：摆动）/ script16（背光，= 我们的 **331**）/ script15（卡面，= **345**）/
    //       script17（暗卡框 —— **不抄，理由见下**）。
    //
    //   **一手 script14 逐句**：
    //     `ins_431(1); randMode(1); type(1); originMode(1);`
    //     `0: scriptNew(script16); scriptNew(script15); scriptNew(script17); pos(0,0,0); layer(16);`
    //     `offset116: rotateTime(30, 9, 0.17453292, 0, 0.17453292);`
    //     `+30: rotateTime(30, 9, -0.17453292, 0, -0.17453292); +30: jmp(offset116, 0); stop();`
    //   ⇒ **摇摆 = ±10°、每 30 帧一个单程**（0.5 秒去、0.5 秒回）—— 这才是 owner 在 B 碎卡上看到的摆动。
    //     ★ 第 21 轮那版抄错了对象：`±2°↔+4° / 120 帧` 是 **script5**（那把卡**弹到屏幕角落**的
    //       "当前卡"演出）的摆动，幅度小、周期长 ⇒ owner「**没有左右摇摆**」；本轮换成 script14 的原值。
    //     判据：B 碎卡自己就摆 ±10°，"各方面 == B 碎卡" ⇒ 同值同周期（不再需要那个 ×2.5 的放大器）。
    //
    //   **Δ1（不写 `ins_431(1)`）**：v8 未知 flag（一手 `engine/anm/th16/07-general.md` §8 标 ❓）——
    //     本仓其它原生移植脚本同款处理（语义未知不发明）。
    //   **Δ2（不写 `originMode(1)`）**：本 VM 的 `entity_pos` 由 C 侧**逐帧**从原生道具坐标
    //     （`item+0xc2c`）写（`ufo.c` 的 `ufo_drops_draw_follow()`，与引擎 `th18:0x446d60` 同一条路）；
    //     那个原点是给**引擎自己 spawn** 的 VM 用的坐标约定，写到我们这支上会让卡整体挪位（实跑验证过）。
    //   **Δ3（不抄 script17 = 暗卡框）**：`abframe.png` 在卡面**下面**（先画），而原生卡图（`_min.png`
    //     的 64×80 画布）**alpha 满 255、四边全不透明**（逐像素量过：5120/5120 个 texel 全覆盖）
    //     ⇒ 卡框**一格都露不出来**，抄了也是死的；少起一支 VM。
    //   **Δ4（背光层 15 / 卡面层 16）**：原生两层都写 `layer(16)`、靠"先 `scriptNew` 的先画"排序；
    //     我们写成 15/16（层号小的先画）拿到**同一个前后关系**，且不依赖创建顺序（见 331 的注）。
    sprite(-1);                          /* 父自己不画（父 VM 只提供位置与摆动）*/
    scriptNew(@SCRIPT_UFO_CARD_GLOW@);   /* ★ 背光层（一条手 script16：加算 + 呼吸）—— 先起 ⇒ 在卡面后 */
    scriptNew(@SCRIPT_UFO_CARD_FACE@);   /* ★ 卡面层（一条手 script15：原生卡图 ×0.3）*/
    type(1);
    layer(16);                           /* 一手 script14：世界层 16（掉落物该在的层）*/
    color(255, 255, 255);
    scale(1.0f, 1.0f);                   /* 父的 scale 传给子（0.3 × 1.0 = 0.3，与一手一致）*/
offset176:
    /* ★ 一手 script14 原值：**+10°** */
    rotateTime(30, 9, 0.17453292f, 0.0f, 0.17453292f);
+30: // 30
    /* ★ 一手 script14 原值：**-10°**（一个来回 = 60 帧 = 1 秒）*/
    rotateTime(30, 9, -0.17453292f, 0.0f, -0.17453292f);
+30: // 60
    jmp(offset176, 0);                   /* 一手 script14：无限摇摆 */
+20: // 80（永不到达；与旧版同形，留作"卡被删掉时脚本的落点"）
    stop();
}
