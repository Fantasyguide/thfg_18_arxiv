script script376 {
    // 109 合成卡「**蓄力攻击 / boss attack 的弹**」的视觉 —— 只给 `assets/sht/append_shooterset.py`
    // 的 `POFV109_BOSSATK_C1..C4` 四组当 shooter 记录的 `+0x22`（**卡侧不 spawn 它**）。
    //
    // ══ owner 口径（2026-10-01，两条）════════════════════════════════════════════════════
    //   ① 「C3 C4 表明我方向上发射 **boss 符卡/非符**，最好**观感一致**」
    //      ⇒ **形状照敌弹**：TH18 的敌弹**本来就是程序化画的** —— 一手 `local/th18.v1.00a/anm/bullet/
    //      bullet.anm.txt` 的 **script2** 就是 `-1: layer(23); resolutionMode(1); sprite(-1);
    //      0: drawPoly(8.0f, 32);` ⇒ 一个 **8 半径的 32 边圆**。本脚本照这个形状（半径 8 = 与
    //      shooter 行的 `hit_w/hit_h = 16` 一致）。
    //   ② ★★ 「只要是**伤害敌人的弹幕**，**除非样式是自机发射的弹幕**，必须上**额外透明度，
    //      50% 起步**」
    //      ⇒ 这颗弹**故意不是自机弹样式**（不是 `pl0X.anm` 那种横条弹），所以**必须**上透明度：
    //      这里 `alpha(112)`（≈ 44% 不透明 = **56% 透明**，过 50% 的线）。
    //      ⇒ 观感一致（形状/配色与 boss 的弹同族） **且** 一眼可辨（半透明 = 这是你打出去的）。
    //
    // ⚠ 注释里**不能写 `@…@` 占位记号**（构建脚本扫全文、不认注释）。
    // ⚠ 脚本里**不写** `scaleTime` / `alphaTime`（会与卡侧每帧的 `set_scale` 打架，见 `sdk.h`）；
    //   常态就 `stop()`，弹的死活由引擎的弹池管（不自己 `delete()`）。
    sprite(-1);
    layer(13);                  // 跟着 282（自机弹那份）的层，别压到敌弹上去
    resolutionMode(1);
    type(1);
    blendMode(1);
    color(255, 255, 255);       // 弹心白
    color2(255, 176, 176);      // 边淡红 —— TH18 敌弹最常见的白/红配
    alpha(112);                 // ★ owner 铁律：50% 起步的额外透明度
    alpha2(96);
    drawPoly(8.0f, 32);         // ★ 与 `bullet.anm` script2 同形状（8 半径 32 边 = 圆）
    stop();
}
