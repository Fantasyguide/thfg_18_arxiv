script script404 {
    // 暴风 · **外层（最外侧 = 深色 ≈ 黑）** —— owner 口径：
    //   「th19 暴风的视觉样子是：**最外侧是深色，大约黑色，中间是红褐色，芯部是透明的**」✓
    //
    // ★ 用**减去混合**（一手 TH19 就是 `blendMode(2)` ✓）反推源色：
    //   源画**白** ⇒ 全减 ⇒ **黑** ✓（这就是最外那圈）
    //   `alpha(0)` 在**内芯** ⇒ 中心**不减** ⇒ **透明** ✓
    //   ⇒ 一条 `drawPoly` 的「两色渐变」本身就给出「**外黑 → 芯透**」✓，
    //     中间的**红褐色**由第二段（script405，暖橙、稍小）叠出来 ✓。
    sprite(-1);
    layer(28);
    blendMode(2);
    drawPoly(1.0f, 24);
    color(255, 255, 255);
    alpha(0);
    color2(255, 255, 255);
    alpha2(200);
    stop();
    interruptLabel(1);
    alphaTime(10, 0, 0);
    alpha2Time(10, 0, 0);
+10:
    delete();
}
