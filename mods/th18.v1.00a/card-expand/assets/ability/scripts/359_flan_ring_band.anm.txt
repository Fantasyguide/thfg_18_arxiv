script script359 {
// 地灵殿（id 111）咲夜蓄力光圈的**外圈粗环** —— 贴图环（`ATLAS_FLAN_RING` / sprite `FLAN_RING`）。
//
// ══ owner 口径（逐字，2026-09-25）══════════════════════════════════════════════════════
//   ·「**我不想要细环，我就想要粗环**。就叠加没问题，**内圈来一个淡的白叠加就行，外圈是相对实一点
//      的白环 or 色环**。」
//   ·「**然后我想的是满充能才是彩虹变色，充能中是白的**」
//   ·（前一条追加）「彩虹的**饱和度降低，明度提高，更透明**」
//   ⇒ 本脚本 = **外圈**：充能中纯白；`interruptLabel(2)`（= 卡侧「蓄满那一帧」发的那个号）之后
//     进 **6 等份色相的循环**：低饱和 + 高明度（V=255、S≈45%）+ 每色 **3 帧**、18 帧一周期。
//   ⚠ 内圈那层淡白在 `script277`（`drawPoly` 实心盘 + 低 alpha），两层同 pos/scale 一起动。
//
// ══ 为什么用贴图而不是 `drawPolyBorder` ═══════════════════════════════════════════════
//   `drawPolyBorder` 的**线宽在局部空间**、不可控（实跑里既细又飘；owner 判定「不要细环」）；
//   贴图环的**粗细就是像素**（20px 带 = 外径的 32%），而且整张图是白的 ⇒ `color` 直接染色。
//   ⇒ 契约：可见外径 = **64 × VM scale**（画布 128、外径 62 取整到 64 好算）⇒ 卡侧 `scale = 半径/64`。
//
// ══ 生命周期（卡侧契约，见 `native/cards/signal.c` 的 `signal_support_flandre()`）══════════
//   · 出生：进入蓄力那一帧 `ce_anm_spawn(..., CE_ANM_ABILITY_SCRIPT_FLAN_RING_BAND, 13)`；
//   · 每帧：`ce_anm_set_pos`（自机位置）+ `ce_anm_set_scale(半径 / 64)`；
//   · 满蓄力那一帧：`ce_anm_interrupt(id, 2)` ⇒ 进彩虹循环；
//   · 收尾（松手 / 被打断 / 删卡）：`ce_anm_interrupt(id, 1)`（20 帧淡出 → 自删）或 `ce_anm_delete`。
    sprite(@FLAN_RING@);
    layer(13);
    zWriteDisable(1);
    blendMode(1);               // 加算叠加（owner：「叠加没问题」）
    color(255, 255, 255);       // 充能中：**白**
    color2(255, 255, 255);      // （两路同写：colorMode 走哪一路都画得出来）
    alpha(150);                 // 「更透明」：从 170 压到 150
    alpha2(150);
    pos(0.0f, 0.0f, 0.0f);      // 归零：位置由卡侧每帧给
    stop();                     // 静态挂住（属性改由中断改），等满蓄力的中断 2
    interruptLabel(1);          // 收尾：20 帧淡出 → 自删
    alphaTime(20, 0, 0);
    alpha2Time(20, 0, 0);
+20: // 20
    delete();
    interruptLabel(2);          // ★ 满蓄力 ⇒ **彩虹变色**（6 等份色相；低饱和、高明度）
offset_rainbow:
    color(255, 150, 150);       // 红   0°（S≈41%、V=255 = 明度高、饱和度低）
    color2(255, 150, 150);
+3: // 3
    color(255, 255, 150);       // 黄  60°
    color2(255, 255, 150);
+3: // 6
    color(150, 255, 150);       // 绿 120°
    color2(150, 255, 150);
+3: // 9
    color(150, 255, 255);       // 青 180°
    color2(150, 255, 255);
+3: // 12
    color(150, 150, 255);       // 蓝 240°
    color2(150, 150, 255);
+3: // 15
    color(255, 150, 255);       // 品红 300°
    color2(255, 150, 255);
+3: // 18
    jmp(offset_rainbow, 0);     // 0 = 无限循环 ⇒ 每 3 帧一色、18 帧一周期
}
