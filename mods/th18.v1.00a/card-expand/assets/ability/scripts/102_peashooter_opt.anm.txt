script script102 {
    // 豌豆射手的子机本体 —— 由 `Player__allocate_option` 起（脚本号当它最后一个栈参）。
    // 骨架照原生子机脚本 script2 / script16 与本 mod 的 script88（破损核心）：
    // interruptLabel(2) 出场、(3) 进店收起、(1) 销毁 —— 这三个都是引擎发的，我们不自己发。
    // 「开大」的视觉不走中断（怕和引擎抢标签），另起一个一次性的 script103 光晕，见 peashooter.c。
    // 贴图是实心的，所以落定后切回普通混合（照 script2），只留一点很慢的呼吸。
-1:
    layer(13);
    blendMode(1);
    type(1);
    zWriteDisable(1);
0:
    interruptLabel(2);
    sprite(@PEASHOOTER_OPT);
    scale(0.0f, 0.0f);
    scaleTime(10, 4, 0.42f, 0.42f);   // 64 px 贴图 → 场上约 27 px
    alpha(255);
+10: // 10
    blendMode(0);
breathe:
    alphaTime(24, 4, 208);
+24: // 34
    alphaTime(24, 4, 255);
+24: // 58
    jmp(breathe, 10);
    stop();
    interruptLabel(3);
    scaleTime(20, 4, 0.0f, 0.0f);
+20:
    stop();
    interruptLabel(1);
    blendMode(1);
    scaleTime(20, 4, 0.0f, 3.0f);
+20:
    delete();
}
