script script123 {
    // 季节槽**外框的一条边**：1×1 的实心矩形模板。
    // ★ 2026-09-10 第三十二轮：TH18 **没有** `drawRectBorder`（opcode 612 用了启动就崩，
    //   见 engine/anm/th18/01-vm-instantiate.md §2b），所以外框改用**四条 1px 矩形拼**出来
    //   —— 这正是 TH16 `front.anm` script114 的 `drawRectBorder(102, 10)` 的等价画法
    //   （102×10 的 1px 描边 = 上/下各 102×1、左/右各 1×8 围住内部 100×8）。
    // C 侧 `season.c` 的 `k_season_frame[]` 逐条设 scale（矩形尺寸）与位置（表面 px）——
    // 上/左各 2px、下/右各 1px（复刻原版「填充在框里差 1px」的错位）；
    // 颜色也由卡每帧写（灰 80 + HUD alpha），脚本里的 alphaTime 只是初始态。
    sprite(-1);
    ins_431(1);
    type(1);
layer(25);   // ★ 2026-09-13：**层 24 → 25**（用户实跑：HUD 挪到 x<257 后整块不见了）。
              //   一手口径（`HUD-LAYOUT.md` §4.1 + `engine/anm/th18/02-render-stages.md`）：
              //   层 24 → 组 37 → 优先级 0x39 → 画进**离屏表面 B**（`0x4ccfa0`）；
              //   层 25 → 组 38 → 优先级 0x3e → 画进 **backbuffer**（`0x4ccfa4`）。
              //   ⇒ 层 24 的内容**只有落在被合成的那块（游玩区矩形）里才看得见** ——
              //   这正是当年把它放 257..365（游玩区左缘内侧）能看见、一挪到原生 HUD 面板带
              //   （x 68..228）就整块消失的原因。本 mod 其余 HUD 全是层 25（112 的 302/303/304/
              //   305/315、107 的 306/307/308、110/111 的 275/276、105/99/136）⇒ 对齐它们。
    resolutionMode(2);
    drawRect(1.0f, 1.0f);   // 1×1；实际尺寸全靠 VM scale（resolutionMode(2) ⇒ 1 单位 = 1 表面 px）
    color(80, 80, 80);
    alpha(255);
+20: // 20
    alphaTime(20, 0, 255);
    stop();
    interruptLabel(4);
    alphaTime(8, 0, 255);    // ★ 2026-09-22 统一动态（TUNING §7.1）：淡回 = **8 帧线性到满亮** —— 一手 = 原生 `abcard.anm` script1–4 的 `interruptLabel(4)`
    caseReturn();
    interruptLabel(5);
    alphaTime(8, 0, 64);     // ★ 2026-09-22 统一动态（TUNING §7.1）：压暗 = **8 帧线性到 64（25%）** —— 一手 = 原生 `abcard.anm` script1–4 的 `interruptLabel(5)`
    caseReturn();
    interruptLabel(1);
    alphaTime(20, 0, 0);
+20: // 40
    delete();
}
