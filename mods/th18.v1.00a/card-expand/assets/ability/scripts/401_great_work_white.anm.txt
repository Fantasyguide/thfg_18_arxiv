script script401 {
    // 伟大之作（id 92）「**全部发亮**」特效 · **第 2 层：平白盘（"黑色变白"那一步）**
    //   AE 配方：「先是**半径增大**，再是**黑色变白**」——半径那半由 397 的径向渐变盘做；
    //   本层是**平白**（`drawPoly(1.0f,32)` 实心盘、加和、`color(255,255,255)`），
    //   几何与 397 完全同步（同样的 42+18 帧扩到 640），**alpha 从 0 渐入到 255**：
    //     · 半径先长（本层几乎不贡献，alpha 还小）；
    //     · 第 30→60 帧 alpha 渐入 ⇒ 加和把**盘内所有像素**一起抬到白 ⇒ 边缘那片"黑"被填成白
    //       （`dst + white×a`）⇒ 观感正是「黑色变白」，第 60 帧（1 s）= 顶点 = **全屏白**。
    //   顶点停留 8 帧后与 397 同步做白色相加的缓出（6 + 30 帧）。α 写 0 会让 VM 不画 ⇒ 用 1 起步。
-1:
    layer(19);
    sprite(-1);
    zWriteDisable(1);
    blendMode(1);
    drawPoly(1.0f, 32);
    color(255, 255, 255);
    alpha(1);
    scale(0.0f, 0.0f);
0:
    scaleTime(42, 0, 180.0f, 180.0f);
    alphaTime(30, 0, 1);                   // 前半：几乎不贡献
+30:
    alphaTime(30, 0, 255);                 // 后半：黑 → 白（第 60 帧到顶）
+30:
    scaleTime(8, 0, 1280.0f, 1280.0f);
+8:
    alphaTime(6, 0, 60);
+6:
    alphaTime(30, 0, 0);
+30:
    delete();
}
