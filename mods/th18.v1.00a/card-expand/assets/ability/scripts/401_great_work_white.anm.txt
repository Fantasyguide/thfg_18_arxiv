script script401 {
    // 伟大之作（id 92）「全部发亮」· **第 2 层：平白盘（"黑色变白"那一步）**
    //
    // AE 配方：「**加和模式**，**中心渐变**，**内白外黑**，先是**半径增大**，再是**黑色变白**」——
    //   第 1 层（397，径向渐变贴图）负责"半径增大"；本层负责"**黑色变白**"。
    //
    // ★★ 关键做法：**用 RGB 从黑升到白**，不碰 alpha。
    //   加和 `blendMode(1)` = `SRCALPHA/ONE` ⇒ `dst + srcColor × srcAlpha`
    //   ⇒ **黑色（RGB 0,0,0）加进去等于没加** ✓（AE 里"黑 = 不发光"同一条）
    //   ⇒ 只要让本层的 RGB 从 **黑** 线性升到 **白**，观感就是"盘内那片黑被一点点提成白" ✓✓
    //   —— 这正是 owner 的原话，也**彻底绕开了 alpha**：
    //   上一版我用 `alpha(1)`/`alphaTime` 控制强度，结果"**一按 C 就全白**" ✗
    //   （图元族吃的是 **color_1 的 A**，而 `alpha()` 写的是 **color_2 的 A** —— 两路寄存器不是同一条；
    //     见 `engine/anm/th18/07-blendmode-rgb-only-alpha-is-sa.md` §2）。
    //
    // 几何与 397 完全同步（慢段 42 帧 + 快段 18 帧，第 60 帧 = 1 s 顶点）；图元的 scale 是**半径**。
    // 顶点后同样做缓出（这里把 RGB 往黑里收，等价于"白色相加的缓出"）。
-1:
    layer(19);
    sprite(-1);
    zWriteDisable(1);
    blendMode(1);
    drawPoly(1.0f, 32);
    color(0, 0, 0);                        /* 黑 = 不贡献（起步绝不白）*/
    alpha(255);                            /* 只写 color_2 的 A；本层不靠它控强度 */
    scale(0.0f, 0.0f);
0:
    scaleTime(42, 0, 100.0f, 100.0f);       /* 慢段：半径 0 → 100（与 397 同步，几乎看不出）*/
    colorTime(30, 0, 0, 0, 0);             /* 前 30 帧仍全黑 */
+30:
    colorTime(30, 0, 255, 255, 255);       /* 后 30 帧：**黑 → 白**（第 60 帧到顶 = 全屏白）*/
+12:
    scaleTime(18, 0, 900.0f, 900.0f);      /* 快段：100 → 900 罩满游玩区（第 60 帧到顶）*/
+18:
    scaleTime(8, 0, 900.0f, 900.0f);       /* 顶点停留 8 帧 */
+8:
    colorTime(6, 0, 80, 80, 80);           /* 缓出：先快落 */
+6:
    colorTime(30, 0, 0, 0, 0);             /* 再长尾收到黑 */
+30:
    delete();
}
