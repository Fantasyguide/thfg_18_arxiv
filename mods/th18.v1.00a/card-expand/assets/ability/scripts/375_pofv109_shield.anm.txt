script script375 {
    // 109 合成卡「**罩子**（抵挡护盾）」的视觉 —— 卡侧**跟随 VM** 画在自机身上。
    //
    // ═══★ 2026-10-01 重写：owner 指认了原版长相 ═══════════════════════════════════════════
    //   owner 原话：「**罩子的指示是一个白色半透明的圆形，外面是黄色的外发光，你找一下**」。
    //   ⇒ 按这句话翻 TH19 的 anm，在 **`th19.dat` 的 `aura.anm`** 里找到一族**按颜色分档**的脚本：
    //
    //   | 脚本 | 配色（`color` / `color2`）| 备注 |
    //   | --- | --- | --- |
    //   | script2  | 白 / 白 | |
    //   | script5  | 白 / `128,128,255`（浅蓝）| |
    //   | script7  | 白 / `128,0,255`（紫）| |
    //   | script9  | 白 / `255,0,0`（红）| |
    //   | **script11** | **白 `255,255,255` / 黄 `255,255,0`** | ★★ **owner 说的就是这个** |
    //   | script12 | 黄 / 黄（加算变体）| 同一族 |
    //   | script16 | 白 | |
    //
    //   ★ 形状靠 **`ins_617(1.0f, 4)`** 这个**双色圆**图元：内层吃 `color/alpha`、外层吃
    //     `color2/alpha2` ⇒ 「**白色半透明的圆 + 黄色外发光**」正是同一条指令的两个颜色槽 ✓。
    //   ⚠ 旧版（本文件的前一稿）从 `effect.anm` script1 抄的是**蓝系** `drawPoly` 画法 —— 那一路是
    //     **反汇编链推出来的**（`FUN_00544dd0` → `push 1`），而 owner 实跑说长相是**白 + 黄** ⇒
    //     以实跑为准，改按 `aura.anm` script11 ✓。
    //
    //   ❓ 自定/近似（如实标注）：
    //     ① script11 里还有 `case(2)/(3)/(4)/(5)` 四段**状态动画**（激活/展开/收缩…）——
    //        本卡的护盾只有「有効 / 无効」两态（`pofv_shield_armed()`）⇒ 这里只取**静止那一态**
    //        的外观，大小交给卡侧每帧的 `ce_anm_set_scale`（**判定 = 绘制**）✓；
    //     ② alpha 取 140/170（一手 script11 在激活态用 `alpha(255)/alpha2(255)`、收缩态用
    //        `alpha(128)/alpha2(128)` ⇒ 本卡取中间，视觉上「白半透明 + 黄外发光」更明显）。
    layer(34);
    sprite(-1);
    anchor(0, 1);
    ins_617(1.0f, 4);              // ★ 一手：双色圆（内 `color/alpha`、外 `color2/alpha2`）
    color(255, 255, 255);          // 内层：白
    alpha(140);                    // **半透明**
    color2(255, 255, 0);           // 外层：黄（外发光）
    alpha2(170);
    stop();
}
