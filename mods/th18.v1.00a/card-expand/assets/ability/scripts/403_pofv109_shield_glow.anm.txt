script script403 {
    // 109 合成卡「**罩子**」的**外层黄色外发光** —— 卡侧第二台跟随 VM（`s_shield_vm2`）。
    //
    // ★★ 2026-10-01（owner：「罩子的指示是一个白色半透明的圆形，**外面是黄色的外发光**」＋「缺少罩子」）：
    //   内层白圆在脚本 375；这一支只画**外面那圈黄光**，与 375 **同坐标、放大一圈**
    //   （卡侧 `POFV_SHIELD_GLOW_SCALE`），且**先起** ⇒ 引擎「后起的画在上面」⇒ 黄光在下 ✓。
    //   ⚠ 为什么拆成两台 VM：TH19 那套用 `ins_617`（双色圆，一条指令两个色槽）表达，
    //     而 **TH18 的 anm 解释器不认得 `ins_617`** ✗ ⇒ 用两台 TH18 原生 `drawPoly` 等价实现 ✓。
    layer(34);
    sprite(-1);
    anchor(0, 1);
    drawPoly(1.0f, 32);
    color(255, 226, 96);           // 黄（外发光）
    alpha(90);
    stop();
}
