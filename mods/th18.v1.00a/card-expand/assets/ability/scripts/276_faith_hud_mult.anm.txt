script script276 {
// ★★ 组合 UI（110 风神录 × 111 地灵殿 共用那一块）—— 行 2 的**倍率字形** VM（`M.MM` 四个字符）。
//
// 贴图：@FONT2_0@ = `ability/font2/FONT2_0.png`（16×32，字形本体 16×20 居中）——
//   即 **TH18 font 2** 的那套数字（`0`..`9`）+ `.`；四个字形由卡侧 `ce_anm_set_sprite`
//   逐字换（`FONT2_0 + d` / `FONT2_DOT`），本脚本只提供**模板**。
//
// ── 为什么不复用现成的 `136_hud_glyph`（`sdk.c` 的 `ce_hud_vm_text_set` 用的那支）────────
//   136 是**共享件**（别的卡也在用它），而它**没有**中断 2/3 这两段：
//     `136: … alpha(255); stop(); interruptLabel(1): delete();`
//   ⇒ 拿它画倍率的话，玩家贴到左下角时**整块都压暗了、只有倍率还亮着**
//     （`cards/faith_hud.c` 的中断是发给整块的）；而且它 `alpha(255)` 固定 ⇒ 连开场那 20 帧
//     淡入都没有，与同块其它元素不同步。所以这里**另起一支**（同字形、同时序），
//     136 一个字节都不动（别人的卡不受影响）。
//
// ── 哪部分是一手、哪部分是 ❓ 自定 ────────────────────────────────────────────────────
//   · 字形 = 一手（TH18 font 2 的原生字形；TH11 那一行的 `"%1.2f"` 用的就是它自己那套 ASCII，
//     本仓没有 TH11 的 ascii 图集 ⇒ 用 TH18 自己的同尺寸字体，见 `04-merged-hud-spec.md` §3.1
//     的宽度预算：7 px 步进 / `.` 4 px 步进 —— 与 TH10 数字位的步进**同值**，所以混排看不出接缝）；
//   · `anchor(1, 1)` = ❓ 自定（理由同 275：纹理是 16×32，墨迹上沿在格内第 6 行 ⇒
//     卡侧摆位时减掉那 3 px（640 空间）——`cards/faith_hud.c` 的 `FAITH_HUD_G_Y_UP`）；
//   · `layer(25)` / `resolutionMode(2)` / `pos` 归零 / 20 帧淡入 / 中断 2/3 = 同 275 的取舍与出处。
//   · `resampleMode(1)` 照抄同族脚本（136 的写法）。
// ⚠ 时间标签：不写 `-1:`（理由同 211/275）。
    sprite(@FONT2_0@);
    pos(0.0f, 0.0f, 0.0f);    // ★ 归零：位置归卡侧 ce_anm_set_pos
    resampleMode(1);
    anchor(1, 1);
    layer(25);
    resolutionMode(2);
    alpha(0);
+20: // 20
    alphaTime(20, 0, 255);
    stop();
    interruptLabel(2);
    alphaTime(30, 0, 255);
    stop();
    // ★★ 2026-09-15f「组合 UI」：**压暗已改由 C 驱动**（`cards/faith_hud.c` 的 `fh_dim_tick()` ——
    //   ❓ 用户口径「UI 统一，不能复刻」：终点 **32**、每帧 **±51**、门 **进 x<−64 ∧ y>400 / 出 y<384 ∨ x>−64**）
    //   ⇒ **本行（中断 3）不再被触发**，保留它只是把**源脚本的一手值**留档：本脚本的时序照 TH11 script32/33/35 抄（64）。
    //   ⚠ 别再发中断 3；两个一手值（TH10 32 / TH11 64）与口径见 `TUNING.md` §2.6f-3 与 `AUDIT` §DC-5。
    interruptLabel(3);
    alphaTime(10, 0, 64);    // ★ 一手值留档（**本行已不被触发** —— 压暗改由 C 驱动，见上）
    stop();
}
